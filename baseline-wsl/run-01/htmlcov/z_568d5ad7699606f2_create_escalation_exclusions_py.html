2Jutda Helpdesk - A Django powered ticket tracker for small enterprise.

5 

6scripts/create_escalation_exclusion.py - Easy way to routinely add particular 

7 days to the list of days on which no 

8 escalation should take place. 

9""" 

10 

11from datetime import timedelta 

12 

13from django.core.management.base import BaseCommand, CommandError 

14from django.utils import timezone 

15 

16from helpdesk.models import EscalationExclusion, Queue 

17 

18day_names = { 

19 "monday": 0, 

20 "tuesday": 1, 

21 "wednesday": 2, 

22 "thursday": 3, 

23 "friday": 4, 

24 "saturday": 5, 

25 "sunday": 6, 

26} 

27 

28 

29class Command(BaseCommand): 

30 def add_arguments(self, parser): 

31 parser.add_argument( 

32 "-d", 

33 "--days", 

34 nargs="*", 

35 choices=list(day_names.keys()), 

36 required=True, 

37 help="Days of week (monday, tuesday, etc). Enter the days as space separated list.", 

38 ) 

39 parser.add_argument( 

40 "-o", 

41 "--occurrences", 

42 default=1, 

43 type=int, 

44 help="Occurrences: How many weeks ahead to exclude this day", 

45 ) 

46 parser.add_argument( 

47 "-q", 

48 "--queues", 

49 nargs="*", 

50 choices=list(Queue.objects.values_list("slug", flat=True)), 

51 help="Queues to include (default: all). Enter the queues slug as space separated list.", 

52 ) 

53 parser.add_argument( 

54 "-x", 

55 "--exclude-verbosely", 

56 action="store_true", 

57 default=False, 

58 help="Display a list of dates excluded", 

59 ) 

60 

61 def handle(self, *args, **options): 

62 days = options["days"] 

63 occurrences = options["occurrences"] 

64 verbose = options["exclude_verbosely"] 

65 queue_slugs = options["queues"] 

66 

67 if not (days and occurrences): 

68 raise CommandError("One or more occurrences must be specified.") 

69 

70 queues = [] 

71 if queue_slugs is not None: 

72 queues = Queue.objects.filter(slug__in=queue_slugs) 

73 

74 for day_name in days: 

75 day = day_names[day_name] 

76 workdate = timezone.now().date() 

77 i = 0 

78 while i < occurrences: 

79 if day == workdate.weekday(): 

80 if EscalationExclusion.objects.filter(date=workdate).count() == 0: 

81 esc = EscalationExclusion.objects.create( 

82 name=f"Auto Exclusion for {day_name}", date=workdate 

83 ) 

84 

85 if verbose: 

86 self.stdout.write( 

87 f"Created exclusion for {day_name} {workdate}" 

88 ) 

89 

90 for q in queues: 

91 esc.queues.add(q) 

92 if verbose: 

93 self.stdout.write(f" - for queue {q}") 

94 

95 i += 1 

96 workdate += timedelta(days=1)
