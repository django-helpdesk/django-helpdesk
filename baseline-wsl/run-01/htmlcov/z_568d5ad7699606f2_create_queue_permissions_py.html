1""" 

2scripts/create_queue_permissions.py - 

3 Create automatically permissions for all Queues. 

4 

5 This is rarely needed. However, one use case is the scenario where the 

6 slugs of the Queues have been changed, and thus the Permission should be 

7 recreated according to the new slugs. 

8 

9 No cleanup of permissions is performed. 

10 

11 It should be safe to call this script multiple times or with partial 

12 existing permissions. 

13""" 

14 

15from django.contrib.auth.models import Permission 

16from django.contrib.contenttypes.models import ContentType 

17from django.core.management.base import BaseCommand 

18from django.db.utils import IntegrityError 

19from django.utils.translation import gettext_lazy as _ 

20 

21from helpdesk.models import Queue 

22 

23 

24class Command(BaseCommand): 

25 def add_arguments(self, parser): 

26 parser.add_argument( 

27 "-q", 

28 "--queues", 

29 nargs="*", 

30 choices=list(Queue.objects.values_list("slug", flat=True)), 

31 help="Queues to include (default: all). Enter the queues slug as space separated list.", 

32 ) 

33 parser.add_argument( 

34 "-x", 

35 "--escalate-verbosely", 

36 action="store_true", 

37 default=False, 

38 help="Display a list of dates excluded", 

39 ) 

40 

41 def handle(self, *args, **options): 

42 queue_slugs = options["queues"] 

43 

44 if queue_slugs is not None: 

45 queues = Queue.objects.filter(slug__in=queue_slugs) 

46 else: 

47 queues = Queue.objects.all() 

48 

49 # Create permissions for the queues, which may be all or not 

50 for q in queues: 

51 self.stdout.write(f"Preparing Queue {q} [{q.slug}]") 

52 

53 if q.permission_name: 

54 self.stdout.write( 

55 f" .. already has `permission_name={q.permission_name}`" 

56 ) 

57 basename = q.permission_name[9:] 

58 else: 

59 basename = q.generate_permission_name() 

60 self.stdout.write( 

61 f" .. generated `permission_name={q.permission_name}`" 

62 ) 

63 q.save() 

64 

65 self.stdout.write(f" .. checking permission codename `{basename}`") 

66 

67 try: 

68 Permission.objects.create( 

69 name=_("Permission for queue: ") + q.title, 

70 content_type=ContentType.objects.get(model="queue"), 

71 codename=basename, 

72 ) 

73 except IntegrityError: 

74 self.stdout.write(" .. permission already existed, skipping")
